### 2.4 Password Grant Token Flow Verify (Smoke Only)

```bash
$ TEST_PASS=$(kubectl --context k3d-test -n platform-test get secret test-personas-perf-auth -o jsonpath='{.data.password}' | base64 -d)
$ curl -sS -X POST "http://127.0.0.1:8082/realms/platform-test/protocol/openid-connect/token" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    --data-urlencode "username=perf-test" \
    --data-urlencode "password=$TEST_PASS" \
    --data-urlencode "grant_type=password" \
    --data-urlencode "client_id=admin-cli" \
    -o /tmp/tokresp.json -w "http_code=%{http_code}\n"
http_code=200

$ python3 -c '
import json
d = json.load(open("/tmp/tokresp.json"))
print(f"keys_present={sorted(d.keys())}")
print(f"access_token_present=True token_type={d[\"token_type\"]} expires_in={d[\"expires_in\"]}")
print(f"access_token_jwt_shape={\"header.payload.signature\" if d[\"access_token\"].count(\".\") == 2 else \"INVALID\"}")
```
